CREATE PROCEDURE dbo.sp_crea_compra
@ID_FACT INT,
@FAR_CODCLIE NUMERIC(8,0),
@mensaje varchar(255) OUTPUT
AS
BEGIN
BEGIN TRY
-- Iniciar la transacción
BEGIN TRANSACTION;
-- Declaración de variables
DECLARE @FAR_TIPMOV INT = 20
DECLARE @FAR_CODCIA CHAR(2) = '25'
DECLARE @FAR_NUMSER CHAR(3)	= '1'
DECLARE @FAR_FBG CHAR(1) = ''
DECLARE @FAR_FECHA DATETIME = CONVERT (date, GETDATE())
DECLARE @FAR_ESTADO CHAR(1) = 'N'
DECLARE @FAR_SIGNO_ARM INT = '1'
DECLARE @FAR_FECHA_PRO DATETIME = CONVERT (date, GETDATE())
DECLARE @FAR_FECHA_COMPRA DATETIME = CONVERT (date, GETDATE())
DECLARE @FAR_CONCEPTO CHAR(60) = 'FACTURA IMPORTADA'
DECLARE @FAR_NUMGUIA NUMERIC(9,0) = '0'
DECLARE @FAR_DIAS INT = (SELECT CLI_AUTO1 FROM DBO.CLIENTES WHERE CLI_CODCLIE = @FAR_CODCLIE AND CLI_CP = 'P')
DECLARE @FAR_STOCK INT = 0
DECLARE @FAR_IMPTO NUMERIC(11,4) = 0
DECLARE @FAR_TOT_DESCTO NUMERIC(11,4) = 0
DECLARE @FAR_DESCTO NUMERIC(11,4) = 0
DECLARE @FAR_GASTOS NUMERIC(11,4) = 0
DECLARE @FAR_MONEDA CHAR(1) = 'S'
DECLARE @ALL_NETO NUMERIC(11,4) 
DECLARE @FAR_NUMFAC NUMERIC(9,0) = (SELECT MAX(ALL_NUMFAC)+1 FROM DBO.allog WHERE ALL_TIPMOV = @FAR_TIPMOV AND ALL_CODCIA = @FAR_CODCIA AND ALL_NUMSER = @FAR_NUMSER)
DECLARE @FAR_NUMOPER INT = (SELECT MAX(ALL_NUMOPER)+1 FROM DBO.ALLOG WHERE ALL_FECHA_DIA=CONVERT (date, GETDATE()))
DECLARE @CAR_NUMDOC NUMERIC(9,0) = (SELECT MAX(CAR_NUMDOC)+1 FROM DBO.CARTERA WHERE CAR_CODCIA = @FAR_CODCIA)
DECLARE @CAA_NOMBRE char(22) = (SELECT SUBSTRING(CLI_NOMBRE, 1, 20) FROM DBO.CLIENTES WHERE CLI_CODCLIE = @FAR_CODCLIE AND CLI_CP = 'P')
DECLARE @CAA_SALDO numeric(11, 2) = (SELECT TOP 1 CAA_SALDO FROM dbo.CARACU WHERE CAA_CODCIA=@FAR_CODCIA AND CAA_CODCLIE=@FAR_CODCLIE AND CAA_CP='P' ORDER BY CAA_FECHA DESC, CAA_NUM_OPER DESC)


--------------------------------------------------
------- CREAR FACTURA EN ALLOG -------------------
--------------------------------------------------
/**/
INSERT INTO dbo.ALLOG(ALL_CODCIA,ALL_FECHA_DIA, ALL_NUMOPER, ALL_CODTRA, ALL_FLAG_EXT, ALL_CODCLIE, ALL_CODART,ALL_IMPORTE_AMORT, 
ALL_IMPORTE, ALL_CHESER, ALL_SECUENCIA, ALL_IMPORTE_DOLL, ALL_CODUSU, ALL_PRECIO, ALL_CODVEN, ALL_FBG, 
ALL_CP, ALL_TIPDOC, ALL_CANTIDAD, ALL_NUMGUIA, ALL_CODBAN, ALL_AUTOCON, ALL_CHENUM, ALL_CHESEC, 
ALL_NUMSER, ALL_NUMFAC, ALL_FECHA_VCTO, ALL_NETO, ALL_BRUTO, 
ALL_GASTOS, ALL_IMPTO, ALL_DESCTO, ALL_MONEDA_CAJA, ALL_MONEDA_CCM, ALL_MONEDA_CLI, ALL_NUMDOC, ALL_LIMCRE_ANT, ALL_LIMCRE_ACT, ALL_TIPO_BLOQ_ANT, ALL_TIPO_BLOQ_ACT, 
ALL_CANT_CHEQ, ALL_NUM_INI, ALL_NUM_OPER2, ALL_SIGNO_ARM, ALL_CODTRA_EXT, ALL_SIGNO_CCM, ALL_SIGNO_CAR, 
ALL_SIGNO_CAJA,ALL_TIPMOV, ALL_NUMSER_C, ALL_NUMFAC_C, ALL_SERDOC, ALL_TIPO_CAMBIO, ALL_FLETE, ALL_SUBTRA, ALL_HORA, 
ALL_FACART, ALL_CONCEPTO, ALL_NUMOPER2, ALL_FECHA_ANT, ALL_SITUACION, ALL_FECHA_SUNAT, ALL_FLAG_SO, ALL_IMPG1, ALL_IMPG2, 
ALL_CTAG1, ALL_CTAG2, ALL_CODSUNAT, ALL_FECHA_PRO, ALL_FECHA_CAN, 
ALL_SERIE_REC, ALL_NUM_RECIBO, ALL_RUC, ALL_CC, ALL_ZONACC, ALL_CODCAJ)
SELECT  @FAR_CODCIA,@FAR_FECHA,@FAR_NUMOPER,1401,'N',IFC.CLI_CODCLI,0,IFC.TOTAL,
0,'i_c',1,0,'ADMIN',0,0,'',
'P','FA',0,0,0,'Compras           :Desct. (%) :0',0,0,
@FAR_NUMSER,@FAR_NUMFAC,IFC.VENCIMIENTO,IFC.TOTAL,IFC.TOTAL,
0,0,0,'S','','S',@CAR_NUMDOC,0,0,NULL,NULL,
0,NULL,0,1,1401,0,1, 0,@FAR_TIPMOV,IFC.ALL_NUMSER,IFC.ALL_NUMFACT,0,3.55,0,'Compras',GETDATE(),
'','Desct. (%) :0',@FAR_NUMOPER,CONVERT(datetime,IFC.FECHA),'',CONVERT(datetime,IFC.FECHA),'',0,0,'','',1,@FAR_FECHA_PRO,CONVERT(datetime,IFC.FECHA),
0,0,IFC.RUC,0,0,0
FROM 
dbo.IMPORT_FACT AS IFC
WHERE
IFC.ID = @ID_FACT AND IFC.ESTADO = 0

SET @ALL_NETO = (SELECT TOTAL FROM dbo.IMPORT_FACT WHERE ID = @ID_FACT AND ESTADO = 0)

--------------------------------------------------
------- CREAR FACTURA EN FACART ------------------
--------------------------------------------------
/**/
INSERT INTO dbo.FACART(FAR_TIPMOV, FAR_CODCIA, FAR_NUMSER, FAR_FBG, FAR_NUMFAC, FAR_NUMSEC, FAR_FECHA, FAR_NUMOPER, FAR_CODCLIE, 
FAR_CODART, FAR_TRANSITO, FAR_ESTADO, FAR_NUMGUIA, FAR_DIAS, FAR_SIGNO_ARM, FAR_PRECIO, FAR_STOCK, FAR_COSPRO, FAR_IMPTO, FAR_TOT_DESCTO, FAR_DESCTO, 
FAR_GASTOS, FAR_BRUTO, FAR_EQUIV, 
FAR_TIPO_CAMBIO,FAR_OTRA_CIA,FAR_NUMSER_C,FAR_NUMFAC_C,FAR_CP,FAR_RUC,FAR_DOCCLI,FAR_DIRCLI,FAR_CLIENTE,
FAR_PRECIO_NETO, FAR_FECHA_COMPRA, FAR_NUM_LOTE, FAR_CANTIDAD, FAR_SIGNO_LOT, FAR_CONCEPTO, FAR_COD_SUNAT,
FAR_DESCRI, FAR_SUBTRA, FAR_CODUSU, FAR_MONEDA, FAR_COSPRO_ANT, FAR_HORA, /**/
FAR_CANTIDAD_P, FAR_NUMOPER2, FAR_FECHA_PRO, FAR_FECHA_CAN, FAR_SUBTOTAL,
FAR_CODLOT, FAR_EX_IGV, FAR_SERGUIA, FAR_ESTADO2, FAR_FLAG_SO,
FAR_SIGNO_CAR,FAR_COSTEO,FAR_COSTEO_REAL,FAR_TIPDOC,FAR_COSPRO_SUP)

SELECT @FAR_TIPMOV,@FAR_CODCIA,@FAR_NUMSER,@FAR_FBG,@FAR_NUMFAC,ROW_NUMBER() OVER( ORDER BY @FAR_TIPMOV),@FAR_FECHA,@FAR_NUMOPER,@FAR_CODCLIE,
TB2.ART_KEY,'',@FAR_ESTADO,@FAR_NUMGUIA,@FAR_DIAS,@FAR_SIGNO_ARM,TB2.TOTAL/TB2.CANTIDAD,ARM_STOCK+(TB2.CANTIDAD*TB2.FAR_EQUIV),ARM_COSPRO,@FAR_IMPTO,@FAR_TOT_DESCTO,@FAR_DESCTO,
@FAR_GASTOS,@ALL_NETO,TB2.FAR_EQUIV,
3.55,'',TB1.ALL_NUMSER,TB1.ALL_NUMFACT,'P',TB1.RUC,'','','',
0,CONVERT(datetime,TB1.FECHA),1,TB2.CANTIDAD*TB2.FAR_EQUIV,0,@FAR_CONCEPTO,'01',
PRE_UNIDAD,Cast(@FAR_CONCEPTO as varchar(20)),'ADMIN',@FAR_MONEDA,ARM_COSPRO,CONVERT(CHAR(8),GETDATE(),108),
TB2.CANTIDAD*TB2.FAR_EQUIV,@FAR_NUMOPER,@FAR_FECHA,@FAR_FECHA,TB2.TOTAL,
'(*)','A',0,'N','K',
1,'A','A','FA',ARM_COSPRO
 
FROM dbo.IMPORT_FACT as TB1
INNER JOIN (SELECT IDFACT,ART_KEY,FAR_EQUIV,SUM(PRECIO) PRECIO,SUM(CANTIDAD) CANTIDAD,SUM(PRECIO*CANTIDAD) TOTAL
FROM dbo.IMPORT_FACT_DET 
WHERE IDFACT = @ID_FACT AND ESTADO = 1
GROUP BY IDFACT,ART_KEY,FAR_EQUIV
) AS TB2 ON(TB1.ID = TB2.IDFACT)
INNER JOIN dbo.ARTICULO AS T2 ON(TB2.ART_KEY=T2.ARM_CODART AND T2.ARM_CODCIA=@FAR_CODCIA)
INNER JOIN dbo.PRECIOS AS T3 ON(TB2.ART_KEY=T3.PRE_CODART AND TB2.FAR_EQUIV=T3.PRE_EQUIV AND T3.PRE_CODCIA=@FAR_CODCIA)
WHERE TB1.ID = @ID_FACT

--------------------------------------------------
------- CREAR FACTURA EN TABESTADOS --------------
--------------------------------------------------
INSERT INTO dbo.TABESTADOS
(TAE_CODCIA,	TAE_TIPMOV,		TAE_NUMSER,		TAE_NUMFAC,		TAE_NUMSEC,							TAE_FECHA_COMPRA,	TAE_CODCLIE,	TAE_PRECIO,
TAE_CANTIDAD,					TAE_EQUIV,		TAE_DESCRI,		TAE_COSPRO,		TAE_COSPRO_ANT,		TAE_CODART,			TAE_ESTADO,		TAE_CODUSU,		TAE_HORA)
SELECT 
@FAR_CODCIA,	@FAR_TIPMOV,	@FAR_NUMSER,	@FAR_NUMFAC,	ROW_NUMBER() OVER( ORDER BY @FAR_TIPMOV),@FAR_FECHA,	@FAR_CODCLIE,	TB2.TOTAL/TB2.CANTIDAD,
TB2.CANTIDAD*TB2.FAR_EQUIV,		TB2.FAR_EQUIV,	PRE_UNIDAD,		0,				0,					TB2.ART_KEY,		'C',			'ADMIN',		CONVERT(CHAR(8),GETDATE(),108)
FROM dbo.IMPORT_FACT as TB1
INNER JOIN (SELECT IDFACT,ART_KEY,FAR_EQUIV,SUM(PRECIO) PRECIO,SUM(CANTIDAD) CANTIDAD,SUM(PRECIO*CANTIDAD) TOTAL
FROM dbo.IMPORT_FACT_DET 
WHERE IDFACT = @ID_FACT AND ESTADO = 1
GROUP BY IDFACT,ART_KEY,FAR_EQUIV
) AS TB2 ON(TB1.ID = TB2.IDFACT)
INNER JOIN dbo.ARTICULO AS T2 ON(TB2.ART_KEY=T2.ARM_CODART AND T2.ARM_CODCIA=@FAR_CODCIA)
INNER JOIN dbo.PRECIOS AS T3 ON(TB2.ART_KEY=T3.PRE_CODART AND TB2.FAR_EQUIV=T3.PRE_EQUIV AND T3.PRE_CODCIA=@FAR_CODCIA)
WHERE TB1.ID = @ID_FACT



--------------------------------------------------
------- CREAR FACTURA EN CARTERA -----------------
--------------------------------------------------
/**/
INSERT INTO DBO.CARTERA(CAR_CP,	CAR_CODCLIE,CAR_CODCIA,CAR_SERDOC,			CAR_NUMDOC,	
CAR_TIPDOC,		CAR_IMPORTE,	CAR_FECHA_INGR,		CAR_FECHA_VCTO,			CAR_NUM_REN,	
CAR_CODART,		CAR_IMP_INI,	CAR_SITUACION,		CAR_NUMSER,				CAR_NUMFAC,
CAR_PRECIO,		CAR_CONCEPTO,	CAR_CODTRA,			CAR_SIGNO_CAR,			CAR_CODVEN,	
CAR_NUMGUIA,	CAR_FBG,		CAR_NOMBRE_BANCO,	CAR_NUM_CHEQUE,			CAR_SIGNO_CAJA,	
CAR_TIPMOV,		CAR_NUMSER_C,	CAR_NUMFAC_C,		CAR_FECHA_VCTO_ORIG,	CAR_COMISION,	
CAR_CODBAN,		CAR_COBRADOR,	CAR_MONEDA,			CAR_SERGUIA,			CAR_FECHA_SUNAT,
CAR_PLACA,		CAR_VOUCHER,	CAR_FLAG_SO	,		CAR_NUMOPER,			CAR_CODUNIBKO)

SELECT 'P',		IFC.CLI_CODCLI,	@FAR_CODCIA,		0,									@CAR_NUMDOC,
'FA',			IFC.TOTAL,	@FAR_FECHA,			CONVERT(datetime,IFC.VENCIMIENTO),	1,
NULL,			IFC.TOTAL,	'',					@FAR_NUMSER,						@FAR_NUMFAC,				
0,				'Desct.(%):0',	1401,				1,									0,					
'0',			'',				'i_nombre_banco',	'i_num_cheque',						0,	
@FAR_TIPMOV,	IFC.ALL_NUMSER,	IFC.ALL_NUMFACT,	CONVERT(datetime,IFC.VENCIMIENTO),	0,				
0,				0,				'S',				0,									CONVERT(datetime,IFC.FECHA),	
NULL,			NULL,			'A',				@FAR_NUMOPER,						''
FROM 
dbo.IMPORT_FACT AS IFC
WHERE
IFC.ID = @ID_FACT AND IFC.ESTADO = 0

--------------------------------------------------
------- CREAR FACTURA EN CARACU ------------------
--------------------------------------------------

INSERT INTO dbo.CARACU 
(CAA_CP,		CAA_CODCLIE,		CAA_CODCIA,			CAA_TIPDOC,	CAA_FECHA,		CAA_NUM_OPER,		CAA_SERDOC,		CAA_NUMDOC,		CAA_IMPORTE,	CAA_SALDO,
CAA_FECHA_VCTO,	CAA_CONCEPTO,		CAA_SIGNO_CAR,		CAA_ESTADO,	CAA_SALDO_CAR,	CAA_TOTAL,			CAA_NUMSER,		CAA_NUMFAC,
CAA_NUMGUIA,	CAA_SIGNO_CAJA,		CAA_TIPMOV,			CAA_FBG,	CAA_HORA,		CAA_CODVEN,			CAA_CODUSU,		CAA_NUMSER_C,	CAA_NUMFAC_C,	CAA_SIGNO_CAJA_REAL,
CAA_NUMPLAN,	CAA_NUM_CHEQUE,		CAA_NOTA,			CAA_SITUACION,CAA_NOMBRE,	CAA_RECIBO,			CAA_INTVEN,		CAA_DIASV,		CAA_DIASA,
CAA_TASAV,		CAA_SERGUIA,		CAA_FECHA_COBRO,	CAA_FLAG_SO,CAA_SERIE,		CAA_TIPO_CAMBIO,	CAA_CODTRA,		CAA_FECHA_PER)

SELECT 'P',		IFC.CLI_CODCLI,		@FAR_CODCIA,		'FA',		@FAR_FECHA,		@FAR_NUMOPER,		0,				@CAR_NUMDOC,	IFC.TOTAL, 	@CAA_SALDO+IFC.TOTAL,
IFC.VENCIMIENTO,'Desct.(%):0',		1,					'N',		IFC.TOTAL,	IFC.TOTAL,		@FAR_NUMSER,	@FAR_NUMFAC,
0,				1,					@FAR_TIPMOV,		'',			GETDATE(),		0,					'ADMIN',		IFC.ALL_NUMSER,	IFC.ALL_NUMFACT,0,
0,				'i_num_cheque',		'',					'',			@CAA_NOMBRE,	0,					0,				0,				0,
0,				0,       CONVERT(datetime,IFC.FECHA),	'A',		0,				3.55,				1401,			CONVERT(datetime,IFC.FECHA)						

FROM 
dbo.IMPORT_FACT AS IFC
WHERE
IFC.ID = @ID_FACT AND IFC.ESTADO = 0


--------------------------------------------------
------- ACTUALIZA STOCK EN ARTICULO --------------
--------------------------------------------------
/**/
UPDATE A
SET
    A.ARM_COSTO_ULT = ISNULL(R.TOTAL / NULLIF(R.CANTIDADES, 0) * FAR_EQUIV, 0),
    A.ARM_COSPRO = ISNULL((R.TOTAL + (A.ARM_STOCK * A.ARM_COSPRO)) / NULLIF((R.CANTIDADES + A.ARM_STOCK), 0), 0),
    A.ARM_STOCK = A.ARM_STOCK + R.CANTIDADES,
    A.ARM_INGRESOS = A.ARM_INGRESOS + R.CANTIDADES,
    A.ARM_SALDO_S = A.ARM_SALDO_S + R.CANTIDADES
FROM dbo.ARTICULO AS A
INNER JOIN (
    SELECT ART_KEY, FAR_EQUIV, SUM(FAR_EQUIV * CANTIDAD) CANTIDADES, SUM(CANTIDAD * PRECIO) TOTAL
    FROM dbo.IMPORT_FACT_DET 
    WHERE IDFACT = @ID_FACT
    GROUP BY ART_KEY, FAR_EQUIV
) R ON (R.ART_KEY = A.ARM_CODART)
WHERE A.ARM_CODCIA = @FAR_CODCIA;

--------------------------------------------------
------- ACTUALIZA STOCK EN LOTE ------------------
--------------------------------------------------

UPDATE L
SET L.LOT_SALDOS = L.LOT_SALDOS + (R.CANTIDAD)
FROM dbo.LOTE AS L
INNER JOIN (SELECT ART_KEY,SUM(CANTIDAD*FAR_EQUIV) CANTIDAD
FROM dbo.IMPORT_FACT_DET 
WHERE IDFACT = @ID_FACT
GROUP BY ART_KEY ) R ON (R.ART_KEY=L.LOT_CODART)
WHERE L.LOT_CODCIA = @FAR_CODCIA
           
-- Insertar en LOTE si el producto no existe
INSERT INTO dbo.LOTE (LOT_CODART, LOT_CODCIA, LOT_SALDOS,LOT_NROLOTE,LOT_CODCLIE)
SELECT R.ART_KEY, @FAR_CODCIA, R.CANTIDAD,'(*)',0
FROM (
    SELECT ART_KEY, SUM(CANTIDAD * FAR_EQUIV) AS CANTIDAD
    FROM dbo.IMPORT_FACT_DET
    WHERE IDFACT = @ID_FACT
    GROUP BY ART_KEY
) AS R
WHERE NOT EXISTS (
    SELECT 1 FROM dbo.LOTE L 
    WHERE L.LOT_CODART = R.ART_KEY AND L.LOT_CODCIA = @FAR_CODCIA
)
  
--------------------------------------------------
------- CAMBIA ESTADO DE FACTURA IMPORTADA -------
--------------------------------------------------

UPDATE dbo.IMPORT_FACT 
SET 
	ESTADO = 1,
    NRO_GUIA = RTRIM(LTRIM(@FAR_NUMSER))+'-'+RTRIM(LTRIM(CONVERT(VARCHAR(30),@FAR_NUMFAC)))
WHERE ID = @ID_FACT 
-- Confirmar la transacción
COMMIT TRANSACTION;

-- Asignar mensaje de salida
SET @mensaje = @FAR_NUMFAC;
END TRY
BEGIN CATCH
-- Si ocurre un error, deshacer todas las operaciones
ROLLBACK TRANSACTION;

-- Capturar el error y asignarlo al mensaje de salida
SET @mensaje = ERROR_MESSAGE();
END CATCH
END

CREATE procedure [dbo].[sp_NotificacionBaja_LE]
	@Codcia CHAR(2),
	@FechaIni DATETIME,
	@FechaFin DATETIME
AS
select 
'INVERSIONES SAN MARTIN S.C.R.L.' AS [RazonSocialEmpresa],
	'20450337839' AS [RUCEmpresa],
	'INVERSIONES SAN MARTIN S.C.R.L.' AS [NombreComercial],
	'220601' [CodigoUbigeo],
	'' [DireccionCompleta],
	'San Martin' [Departamento],
	'Mariscal Cáceres' [Provincia],
	'Juanjuí' [Distrito],
	'PE' [Pais],
	--DATEADD(DAY,1,ALL_FECHA_DIA) [FechaBaja],
	ALL_FECHA_DIA [FechaBaja],
	(select top 1 ALL_FECHA_DIA from ALLOG l1 where l1.ALL_CODCIA=t1.ALL_CODCIA and l1.ALL_TIPMOV=t1.ALL_TIPMOV and l1.ALL_FBG=t1.ALL_FBG and l1.ALL_NUMSER=t1.ALL_NUMSER and l1.ALL_NUMFAC=t1.ALL_NUMFAC and ALL_CODTRA<>1111) [FechaEmisiónDocumentoAnulado],
	case all_tipmov when 10 then
			case when all_fbg ='F' then '01' else '03' end 
		when 97 then '07'
		when 98 then '08'
		else ''
	end [TipoDocumentoAnulado],
	case all_tipmov when 10 then
		CASE WHEN all_fbg='F' THEN 'FA' + RIGHT('000'+ ltrim(str(all_numser)),2) ELSE 'BO' + RIGHT('000'+ ltrim(str(all_numser)),2) END
		when 97 then 
			CASE WHEN T1.ALL_TIPO_BLOQ_ANT ='F' THEN 'FA' + RIGHT('000'+ ltrim(str(all_numser)),2) WHEN T1.ALL_TIPO_BLOQ_ANT ='F' THEN  'BO'+ RIGHT('000'+ ltrim(str(all_numser)),2) ELSE '' END 
		when 98 then 
			CASE WHEN T1.ALL_TIPO_BLOQ_ANT ='F' THEN 'FA' + RIGHT('000'+ ltrim(str(all_numser)),2) WHEN T1.ALL_TIPO_BLOQ_ANT ='F' THEN  'BO'+ RIGHT('000'+ ltrim(str(all_numser)),2) ELSE '' END 
		else ''
	end [SerieDocumentoAnulado],
	--CASE WHEN CAST(ALL_NUMFAC AS NVARCHAR(MAX)) IN ('22143','22144','22145','22146') THEN 'FF11' ELSE 'FF12' END [SerieDocumentoAnulado],
	right('00000000'+ltrim(CAST(ALL_NUMFAC AS NVARCHAR(MAX))),8) [CorrelativoDocumentoAnulado],
	'ERROR INTERNO' [MotivoBaja],
	--(SELECT COUNT(*) + 1 FROM DeclaracionDocumentoElectronico WHERE Fecha = CAST(GETDATE() AS date) AND TipoDocumento = 'RA') [Correlativo],
	(SELECT COUNT(*) + 3 FROM DeclaracionDocumentoElectronico WHERE Fecha = ALL_FECHA_DIA AND TipoDocumento = 'RA') [Correlativo],
	--CONVERT(nvarchar(100),all_fecha_dia,12) + ltrim(str(all_numoper)) [Correlativo],
	CONVERT(nvarchar(100),all_fecha_dia,12) + ltrim(str(all_numoper)) [DocEntry]
FROM ALLOG T1 INNER JOIN PARGEN T2 ON T1.all_CODCIA=T2.PAR_CODCIA 
WHERE all_codcia=@Codcia AND all_tipmov in (10,97,98) and ALL_CODTRA=1111  AND all_fecha_dia >=@FechaIni AND all_fecha_dia <=@FechaFin and ALL_CP='C' AND ALL_DOC_ELECTRONICO='A' and isnull(t1.ALL_ESTADO_FE,0) in (0,3)
and case all_tipmov when 10 then ALL_FBG else ALL_TIPO_BLOQ_ANT end = 'F'
ORDER BY all_fecha_dia,all_fbg,all_numser,all_numfac
GO



SET ANSI_NULLS ON
SET QUOTED_IDENTIFIER ON
GO

CREATE PROCEDURE [dbo].[usp_GetResumenDocumento2]
	@Cia CHAR(2),
	@FechaIni DATETIME,
	@FechaFin DATETIME
AS
/* BOLETAS */
SELECT distinct  
	CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER)) [NumOper]
	,CONVERT(numeric(18,4),CASE WHEN FAR_EX_IGV ='A' THEN FAR_BRUTO ELSE 
		( CONVERT(NUMERIC(13,4),FAR_BRUTO )) END) [LineTotal]
	,convert(numeric(18,4),CASE WHEN FAR_EX_IGV ='A'  THEN 0 else
		FAR_IMPTO END) [IGV]
	,CONVERT(numeric(18,2),FAR_BRUTO + FAR_IMPTO) [SubTotal]
	INTO #Ventas	
FROM facart 
WHERE far_codcia=@Cia AND far_tipmov in (10) AND far_fecha_compra >=@FechaIni AND far_fecha_compra <=@FechaFin
AND far_flag_so<>'X' and FAR_FBG ='B' and FAR_CP='C'


SELECT distinct  
	CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER)) [NumOper]
	,sum(CONVERT(numeric(18,2),FAR_PRECIO_NETO)) [PrecioMin]
	INTO #Ventas2
FROM facart 
WHERE far_codcia=@Cia AND far_tipmov =10 AND far_fecha_compra >=@FechaIni AND far_fecha_compra <=@FechaFin
AND far_flag_so<>'X' and FAR_FBG='B' and FAR_CP='C'
GROUP BY CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER))
/* ====================== */

/* NOTA CREDITO */
select * 	INTO #NotaCred	from (
SELECT distinct 
	CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER)) [NumOper]
	,FAR_OC
	,FAR_TIPO_BLOQ_ACT1
	,FAR_NUMSER_C
	,FAR_NUMFAC_C 
	,CONVERT(numeric(18,4),CASE WHEN FAR_EX_IGV ='A' THEN FAR_BRUTO ELSE 
		( CONVERT(NUMERIC(13,4),FAR_BRUTO )) END) [LineTotal]
	,convert(numeric(18,4),CASE WHEN FAR_EX_IGV ='A'  THEN 0 else
		FAR_IMPTO END) [IGV]
	,CONVERT(numeric(18,2),FAR_BRUTO + FAR_IMPTO) [SubTotal]
	

FROM facart 
WHERE far_codcia=@Cia AND far_tipmov=97 AND far_fecha_compra >=@FechaIni AND far_fecha_compra <=@FechaFin AND far_cp='C'
AND far_codart<>0 and FAR_ESTADO2 <> 'L'
union all
SELECT distinct 
	CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER)) [NumOper]
	,FAR_OC
	,FAR_TIPO_BLOQ_ACT1
	,FAR_NUMSER_C
	,FAR_NUMFAC_C 
	,CONVERT(numeric(18,4),CASE WHEN FAR_EX_IGV ='A' THEN FAR_SUBTOTAL ELSE 
		( CONVERT(NUMERIC(13,4),FAR_BRUTO )) END) [LineTotal]
	,convert(numeric(18,4),CASE WHEN FAR_EX_IGV ='A'  THEN 0 else
		FAR_IMPTO END) [IGV]
	,CONVERT(numeric(18,2),FAR_BRUTO + FAR_IMPTO) [SubTotal]
	

FROM facart 
WHERE far_codcia=@Cia AND far_tipmov=97 AND far_fecha_compra >=@FechaIni AND far_fecha_compra <=@FechaFin AND far_cp='C'
AND far_codart=0
) tab
/* ====================== */


/*  NOTA DE DEBITO  */
select * 	INTO #NotaDEB	from (
SELECT
	CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER)) [NumOper]
	,FAR_OC
	,FAR_TIPO_BLOQ_ACT1
	,FAR_NUMSER_C
	,FAR_NUMFAC_C 
	,sum(CONVERT(numeric(18,4),	CASE WHEN FAR_EX_IGV ='A' THEN (CONVERT(NUMERIC(13,4),(FAR_CANTIDAD / FAR_EQUIV)) * FAR_PRECIO) ELSE 
		(CONVERT(NUMERIC(13,4),(FAR_CANTIDAD / FAR_EQUIV)) * CONVERT(NUMERIC(13,4),(FAR_PRECIO / 1.18))) END)) [LineTotal]
	,SUM(convert(numeric(18,4),CASE WHEN FAR_EX_IGV ='A' THEN 0 else
		(FAR_PRECIO * CONVERT(NUMERIC(13,4),(FAR_CANTIDAD / FAR_EQUIV))) - (CONVERT(NUMERIC(13,4),(FAR_PRECIO / 1.18)) * CONVERT(NUMERIC(13,4),(FAR_CANTIDAD / FAR_EQUIV))) END)) [IGV]
	,SUM(CONVERT(numeric(18,2),FAR_PRECIO * CONVERT(NUMERIC(13,4),(FAR_CANTIDAD / FAR_EQUIV)))) [SubTotal]
	

FROM facart 
WHERE far_codcia=@Cia AND far_tipmov=98 AND far_fecha_compra >=@FechaIni AND far_fecha_compra <=@FechaFin AND far_cp='C'
AND far_codart<>0 and FAR_ESTADO2 <> 'L'
GROUP BY CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER))	,FAR_OC	,FAR_TIPO_BLOQ_ACT1	,FAR_NUMSER_C	,FAR_NUMFAC_C 
union all
SELECT
	CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER)) [NumOper]
	,FAR_OC
	,FAR_TIPO_BLOQ_ACT1
	,FAR_NUMSER_C
	,FAR_NUMFAC_C 
	,sum(CONVERT(numeric(18,4),	CASE WHEN FAR_EX_IGV ='A' THEN FAR_SUBTOTAL ELSE 
		(CONVERT(NUMERIC(13,4),(FAR_SUBTOTAL / 1.18))) END)) [LineTotal]
	,SUM(convert(numeric(18,4),CASE WHEN FAR_EX_IGV ='A' THEN 0 else
		(FAR_SUBTOTAL - (CONVERT(NUMERIC(13,4),(FAR_SUBTOTAL / 1.18)) )) END)) [IGV]
	,SUM(CONVERT(numeric(18,2),FAR_SUBTOTAL )) [SubTotal]
	

FROM facart 
WHERE far_codcia=@Cia AND far_tipmov=98 AND far_fecha_compra >=@FechaIni AND far_fecha_compra <=@FechaFin AND far_cp='C'
AND far_codart=0
GROUP BY CONVERT(nvarchar(100),FAR_FECHA,12) + ltrim(str(FAR_NUMOPER))	,FAR_OC	,FAR_TIPO_BLOQ_ACT1	,FAR_NUMSER_C	,FAR_NUMFAC_C 
) tab
/* FIN NOTA DEBITO   */

		,@vchTicket VARCHAR(20)
		,@intCodigoEstado INT 
		,@vchObservacion VARCHAR(MAX)
	)
AS
BEGIN
	INSERT INTO DeclaracionDocumentoElectronico
	(
		TipoDocumento,
		NroOficial,
		Ticket,
		CodigoEstado,
		Observacion,
		Fecha
	)

	(
		isnull(@chrTipoDocumento,CASE WHEN @vchObservacion LIKE '%FACTURA%' THEN '01' WHEN @vchObservacion LIKE '%BOLETA%' THEN '03' WHEN @vchObservacion LIKE '%CREDITO%' THEN '07' WHEN @vchObservacion LIKE '%DEBITO%' THEN '08' WHEN @vchObservacion LIKE '%BAJA%' THEN 'RA' ELSE '' END), 
		@vchNroOficial, 
		@vchTicket, 
		@intCodigoEstado,
		@vchObservacion, 
		CONVERT(date,GETDATE())
	)
	
	SET @intId = @@identity
	
END
